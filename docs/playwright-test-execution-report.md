test('T5.5.1 - 创建寄存器后设备树状态保持', async ({ page }) => {
  // 1. 导航到页面
  await page.goto('http://localhost:3002');

  // 2. 展开连接和从站
  await page.click('[data-type="connection"]');
  await page.click('[data-type="slave"]');

  // 3. 打开寄存器标签页
  await page.click('text=保持寄存器');

  // 4. 创建寄存器
  await page.click('button:has-text("添加寄存器")');
  await page.fill('#regStartAddr', '40002');
  await page.fill('#regHexData', '00C8');
  await page.click('button:has-text("保存")');

  // 5. 验证状态保持 (关键!)
  await expect(page.locator('[data-type="connection"] .tree-expand'))
    .toHaveText('▼');  // 连接应该保持展开
  await expect(page.locator('[data-type="slave"] .tree-expand'))
    .toHaveText('▼');  // 从站应该保持展开

  // 6. 验证数据正确
  await expect(page.locator('td')).toContainText(['40002', '00C8', '200.000']);
});
